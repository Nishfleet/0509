### Queries and raw responses

Each dataset below is one GraphQL request, run once for the window. The R2 and Workflow responses keep only this app's buckets and production Workflows; the account's other buckets belong to other products and are not quoted here.

```graphql
workersInvocationsAdaptive: query { viewer { accounts(filter: { accountTag: "<accountTag>" }) { workersInvocationsAdaptive(limit: 10000, filter: { scriptName: "0509", datetime_geq: "2026-10-29T00:00:00Z", datetime_lt: "2026-11-05T00:00:00Z" }, orderBy: [date_ASC]) { dimensions { date isPreview usageModel } sum { requests errors cpuTimeUs } } } } }
d1AnalyticsAdaptiveGroups: query { viewer { accounts(filter: { accountTag: "<accountTag>" }) { d1AnalyticsAdaptiveGroups(limit: 10000, filter: { databaseId: "746c6e3d-782e-443a-82d6-28ca93a16294", datetime_geq: "2026-10-29T00:00:00Z", datetime_lt: "2026-11-05T00:00:00Z" }, orderBy: [date_ASC]) { dimensions { date } sum { rowsRead rowsWritten } } } } }
r2OperationsAdaptiveGroups: query { viewer { accounts(filter: { accountTag: "<accountTag>" }) { r2OperationsAdaptiveGroups(limit: 10000, filter: { datetime_geq: "2026-10-29T00:00:00Z", datetime_lt: "2026-11-05T00:00:00Z" }) { dimensions { bucketName actionType date } sum { requests } } } } }
r2StorageAdaptiveGroups: query { viewer { accounts(filter: { accountTag: "<accountTag>" }) { r2StorageAdaptiveGroups(limit: 10000, filter: { datetime_geq: "2026-10-29T00:00:00Z", datetime_lt: "2026-11-05T00:00:00Z" }) { dimensions { bucketName date } max { payloadSize metadataSize objectCount } } } } }
workflowsAdaptiveGroups: query { viewer { accounts(filter: { accountTag: "<accountTag>" }) { workflowsAdaptiveGroups(limit: 10000, filter: { datetime_geq: "2026-10-29T00:00:00Z", datetime_lt: "2026-11-05T00:00:00Z" }) { dimensions { workflowName date } sum { stepCount } } } } }

queueMessageOperationsAdaptiveGroups: query { viewer { accounts(filter: { accountTag: "<accountTag>" }) { queueMessageOperationsAdaptiveGroups(limit: 10000, filter: { datetime_geq: "2026-10-29T00:00:00Z", datetime_lt: "2026-11-05T00:00:00Z" }) { dimensions { queueName date } sum { billableOperations } } } } }
```
